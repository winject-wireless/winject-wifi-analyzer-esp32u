cmake_minimum_required(VERSION 3.16)
project(winject_wifi_analyzer_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(ZLIB REQUIRED)
add_executable(winject-wifi-analyzer-cpp host_server.cpp wjiq.cpp ofdm.cpp mpdu.cpp dsss_decoder.cpp)
target_compile_options(winject-wifi-analyzer-cpp PRIVATE -O3)
target_link_libraries(winject-wifi-analyzer-cpp PRIVATE Threads::Threads OpenSSL::Crypto ZLIB::ZLIB)
enable_testing()
add_executable(test-simd tests/test_simd.cpp)
target_compile_options(test-simd PRIVATE -UNDEBUG)
add_test(NAME simd COMMAND test-simd)
add_executable(test-mpdu tests/test_mpdu.cpp mpdu.cpp)
target_compile_options(test-mpdu PRIVATE -UNDEBUG)
target_link_libraries(test-mpdu PRIVATE ZLIB::ZLIB)
add_test(NAME mpdu COMMAND test-mpdu)
add_executable(test-dsss ../tests/test_dsss_native.cpp dsss_decoder.cpp)
target_include_directories(test-dsss PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(test-dsss PRIVATE ZLIB::ZLIB)
add_test(NAME dsss COMMAND test-dsss)
add_executable(test-snr tests/test_snr.cpp)
target_include_directories(test-snr PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(test-snr PRIVATE -UNDEBUG)
add_test(NAME snr COMMAND test-snr)
set_tests_properties(simd mpdu dsss snr PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/..)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME server-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_server_integration.py $<TARGET_FILE:winject-wifi-analyzer-cpp>)
  set_tests_properties(server-integration PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/.. TIMEOUT 25)
endif()
